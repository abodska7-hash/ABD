<script>
  const STORAGE_KEY = 'spices_factory_orders_v1';
  let orders = JSON.parse(localStorage.getItem(STORAGE_KEY) || '[]');

  const stages = [
    { key: 'commercial', label: '1. استلام تجاري', class: 'st-commercial', next: 'طابق الإنتاج' },
    { key: 'production', label: '2. قيد الإنتاج والخلط', class: 'st-production', next: 'طابق التعبئة' },
    { key: 'packaging', label: '3. قيد التعبئة والفرز', class: 'st-packaging', next: 'تسليم للتجاري' },
    { key: 'done', label: '4. جاهز / تم التسليم', class: 'st-done', next: 'إعادة تفعيل' }
  ];

  function save() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(orders));
    render();
  }

  document.getElementById('orderForm').addEventListener('submit', (e) => {
    e.preventDefault();
    const newOrder = {
      id: Date.now().toString().slice(-4),
      client: document.getElementById('client').value,
      category: document.getElementById('category').value,
      details: document.getElementById('details').value,
      quantity: document.getElementById('quantity').value,
      packaging: document.getElementById('packaging').value,
      shortage: document.getElementById('shortage').value.trim(),
      stageIndex: 1 // يُرسل مباشرة لطابق الإنتاج
    };
    orders.unshift(newOrder);
    save();
    document.getElementById('orderForm').reset();
  });

  function nextStage(id) {
    const o = orders.find(x => x.id === id);
    if (!o) return;
    o.stageIndex = (o.stageIndex + 1) % stages.length;
    save();
  }

  function toggleShortage(id) {
    const o = orders.find(x => x.id === id);
    if (!o) return;
    const current = o.shortage;
    const update = prompt("تحديث النواقص المطلوبة من القسم التجاري:", current || "");
    if (update !== null) {
      o.shortage = update.trim();
      save();
    }
  }

  function deleteOrder(id) {
    if (confirm("هل تريد حذف هذه الطلبية نهائياً؟")) {
      orders = orders.filter(x => x.id !== id);
      save();
    }
  }

  function render() {
    const tbody = document.getElementById('ordersTable');
    tbody.innerHTML = '';
    orders.forEach(o => {
      const stage = stages[o.stageIndex] || stages[0];
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td><strong>#${o.id}</strong></td>
        <td>${o.client}</td>
        <td><strong>${o.details}</strong><br><small style="color:#64748b;">${o.category} - ${o.quantity}</small></td>
        <td>${o.packaging}</td>
        <td><span class="badge ${stage.class}">${stage.label}</span></td>
        <td>${o.shortage ? `<span class="shortage-tag">${o.shortage}</span>` : '<span style="color:#10b981; font-size:11px;">مكتمل المواد</span>'}</td>
        <td>
          <button class="btn-step" onclick="nextStage('${o.id}')">نقل: ${stage.next}</button>
          <button class="btn-shortage" onclick="toggleShortage('${o.id}')">نقص</button>
          <button class="btn-del" onclick="deleteOrder('${o.id}')">حذف</button>
        </td>
      `;
      tbody.appendChild(tr);
    });
  }

  render();
</script>
